  //%attributes = {"invisible":true,"preemptive":"capable","folder":"Utility","lang":"en"} comment added and reserved by 4D.
  // Folder_DoesExist (path to folder) : doesExist
  // 
  // DESCRIPTION:
  //   Returns true if the folder exists
  //
C_TEXT($1;$Folder_vt_fullPath)  // path to folder
C_BOOLEAN($0;$Folder_vb_doesExist)
  // ----------------------------------------------------
  // MODIFICATION HISTORY:
  //   Added: DB (7/17/03 @ 15:46:39)
  // ----------------------------------------------------

$Folder_vb_doesExist:=False
If (Asserted(Count parameters=1))
$Folder_vt_fullPath:=$1

If ($Folder_vt_fullPath#"")
If (Test path name($Folder_vt_fullPath)=Is a folder)
$Folder_vb_doesExist:=True
End if 
End if 
End if   // ASSERT

$0:=$Folder_vb_doesExist
