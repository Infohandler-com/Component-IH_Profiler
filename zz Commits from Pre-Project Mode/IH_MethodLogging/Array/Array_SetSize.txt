  //%attributes = {"invisible":true,"preemptive":"capable","folder":"Utility","lang":"en"} comment added and reserved by 4D.
  // (PM) Array_SetSize
  // Changes the size of one or multiple arrays
  // $1 = Size
  // $2 etc. = Pointers to arrays

C_LONGINT($1;$size;$param;$currentSize)
C_POINTER(${2};$array)

OnErr_Install_Handler ("OnErr_GENERIC")

$size:=$1
If ($size<0)
$size:=0
End if 

For ($param;2;Count parameters)

$array:=${$param}
$currentSize:=Size of array($array->)
Case of 
: ($currentSize<$size)
INSERT IN ARRAY($array->;$currentSize+1;$size-$currentSize)

: ($currentSize>$size)
DELETE FROM ARRAY($array->;$size+1;$currentSize-$size)
End case 

End for 

OnErr_Install_Handler ("OnErr_GENERIC")
