  //%attributes = {"invisible":true,"shared":true,"preemptive":"capable","lang":"en"} comment added and reserved by 4D.
  // Profiler_GetLocalProfileStats () : profileStatsAsText
  // Profiler_GetLocalProfileStats () : text
  // 
  // DESCRIPTION
  //   This method returns the contents of
  //   the profile arrays in a standard format.
  //   Only the stats for the current process are returned.
  //
C_TEXT($0;$vt_profileStatsAsText)
  // ----------------------------------------------------
  // HISTORY
  //   Created by: DB (05/24/2017)
  // ----------------------------------------------------

Init_ThreadSafe 
$vt_profileStatsAsText:=""

If (Size of array(_ML_Flame_Method)>0)
$vt_profileStatsAsText:="\r"+("-"*40)+"\r"
$vt_profileStatsAsText:=$vt_profileStatsAsText+("-"*40)+"\r"
$vt_profileStatsAsText:=$vt_profileStatsAsText+"METHOD NAME (LOCAL)\t"
$vt_profileStatsAsText:=$vt_profileStatsAsText+"Call Count\t"
$vt_profileStatsAsText:=$vt_profileStatsAsText+"Min (ms)\t"
$vt_profileStatsAsText:=$vt_profileStatsAsText+"Avg (ms)\t"
$vt_profileStatsAsText:=$vt_profileStatsAsText+"Max (ms)\t"
$vt_profileStatsAsText:=$vt_profileStatsAsText+"Total (ms)\r"
$vt_profileStatsAsText:=$vt_profileStatsAsText+("-"*40)+"\r"

SORT ARRAY(_ML_Perf_Method;_ML_Perf_minTime;_ML_Perf_maxTime;_ML_Perf_totalTime;_ML_Perf_count)
For ($i;1;Size of array(_ML_Perf_Method))
$vt_profileStatsAsText:=$vt_profileStatsAsText+_ML_Perf_Method{$i}+Char(Tab)
$vt_profileStatsAsText:=$vt_profileStatsAsText+String(_ML_Perf_count{$i})+Char(Tab)
$vt_profileStatsAsText:=$vt_profileStatsAsText+String(_ML_Perf_minTime{$i})+Char(Tab)
If (_ML_Perf_totalTime{$i}>0)
$vt_profileStatsAsText:=$vt_profileStatsAsText+String(_ML_Perf_totalTime{$i}/_ML_Perf_count{$i};"###,###,###,###,##0.00")+Char(Tab)
Else 
$vt_profileStatsAsText:=$vt_profileStatsAsText+"0"+Char(Tab)
End if 
$vt_profileStatsAsText:=$vt_profileStatsAsText+String(_ML_Perf_maxTime{$i})+Char(Tab)
$vt_profileStatsAsText:=$vt_profileStatsAsText+String(_ML_Perf_totalTime{$i})+Char(Carriage return)  //   Mod: DB (01/28/2014)
End for 
$vt_profileStatsAsText:=$vt_profileStatsAsText+("-"*40)+"\r"
End if 

$0:=$vt_profileStatsAsText
