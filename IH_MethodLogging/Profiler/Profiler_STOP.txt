  //%attributes = {"invisible":true,"shared":true,"preemptive":"capable","folder":"Profiler Shared Methods","lang":"en"} comment added and reserved by 4D.
  // Profiler_STOP (methodName {;optionalText})
  // Profiler_STOP (text {;text})
  // 
  // DESCRIPTION
  //   Used to capture the end of a segment of code that
  //   is to be monitored.
  //   NOTE: Spaces in $1 & $2 are replaced by "_".
  //
C_TEXT($1;$nameToTrack)  // method name
C_TEXT($2;$vt_extraText)  // Optional stuff
  // ----------------------------------------------------
  // HISTORY
  //   Created by: DB (11/23/07)
  //   Mod: DB (10/26/2010) - Make sure that we get the method name that we are expecting
  //   Mod: DB (11/22/2010) - support performance tracking switch
  //   Mod: DB (01/20/2011) - Improved Performance of this method, significantly faster
  //   Mod: DB (01/28/2014) - Add tracking globally
  //   Mod: DB (05/22/2017) - Moved to v16 component
  // ----------------------------------------------------

ASSERT(Count parameters>0)
ASSERT(Count parameters<=2)

C_TEXT($vt_FullMethodName)
$nameToTrack:=Replace string($1;" ";"_")
If (Count parameters>=2)
$vt_extraText:=Replace string($2;" ";"_")
End if 
$vt_FullMethodName:=$nameToTrack
If ($vt_extraText#"")
$vt_FullMethodName:=$vt_FullMethodName+"("+$vt_extraText+")"
End if 


Init_ThreadSafe 

C_LONGINT($size)
$size:=Size of array(_ML_STACK_CallingStack)

If ($size>0)

  // # Make sure that the STOPPED method matches what is expected
If ($nameToTrack#_ML_STACK_CallingStack{$size}) | ($vt_extraText#_ML_STACK_ExtraText{$size})

  //   Mod: DB (01/28/2014) - Improved error checks
C_TEXT($vt_msg)
$vt_msg:=" called with $1 = '"+$nameToTrack+"'"
If ($vt_extraText#"")
$vt_msg:=$vt_msg+" and $2 = '"+$vt_extraText+"'"
End if 
$vt_msg:=$vt_msg+" but expecting $1 = '"+_ML_STACK_CallingStack{$size}+"'"
If (_ML_STACK_ExtraText{$size}#"")
$vt_msg:=$vt_msg+" and $2 = '"+_ML_STACK_ExtraText{$size}+"'"
End if 
Log_ERR (Current method name+$vt_msg;"Process")

End if 


C_REAL($vr_msStop)
$vr_msStop:=Milliseconds

  // # Track the time
C_LONGINT($vl_timeToExecute)
$vl_timeToExecute:=$vr_msStop-_ML_STACK_msStart{$size}  // time between start & end
$vl_timeToExecute:=$vl_timeToExecute-_ML_STACK_wasteTime{$size}  // remove the child method's time
If ($size>1)  // add our execute time to the calling parent method
_ML_STACK_wasteTime{$size-1}:=_ML_STACK_wasteTime{$size-1}+_ML_STACK_wasteTime{$size}  // add child method times
_ML_STACK_wasteTime{$size-1}:=_ML_STACK_wasteTime{$size-1}+$vl_timeToExecute  // add current method times
End if 

  // Track by method name
Performance_TrackCall ($vt_FullMethodName;$vl_timeToExecute;->_ML_Perf_Method;->_ML_Perf_minTime;->_ML_Perf_maxTime;->_ML_Perf_totalTime;->_ML_Perf_count)

  // Track by flamegraph name
C_TEXT($vt_MethodName_FlameGraph)
$vt_MethodName_FlameGraph:=STR_GetFlameGraphNameFromStack ($vt_FullMethodName;->_ML_STACK_CallingStack)
Performance_TrackCall ($vt_MethodName_FlameGraph;$vl_timeToExecute;->_ML_Flame_Method;->_ML_Flame_minTime;->_ML_Flame_maxTime;->_ML_Flame_totalTime;->_ML_Flame_count)


  // Pull it off the list
Array_SetSize ($size-1;->_ML_STACK_CallingStack;->_ML_STACK_ExtraText;->_ML_STACK_msStart;->_ML_STACK_wasteTime)
End if 

  // Deal with tab levels
__incrementLevel:=__incrementLevel-1
If (__incrementLevel<0)
Log_ERR (Current method name+" has and __incrementLevel of "+String(__incrementLevel);"Process")
__incrementLevel:=0
End if 

  // If this STOP is the last one from the stack, push to global tracking
If (Size of array(_ML_STACK_CallingStack)=0) & (Size of array(_ML_Perf_Method)>0)
Performance_PushToGlobal_Perf 
Performance_PushToGlobal_Flame 
End if
