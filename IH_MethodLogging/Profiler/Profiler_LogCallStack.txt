  //%attributes = {"invisible":true,"shared":true,"preemptive":"capable","folder":"Profiler Shared Methods","lang":"en"} comment added and reserved by 4D.
  // Profiler_LogCallStack ({errorNo})
  // 
  // DESCRIPTION
  //   Dumps the local process's call stack to the log.
  //   Appends the errorNo if one is passed in.
  //
C_LONGINT($1;$err)
  // ----------------------------------------------------
  // HISTORY
  //   Created by: DB (01/14/09)
  //   Mod: DB (05/22/2017) - Moved to v16 component
  // ----------------------------------------------------

Init_ThreadSafe 

If (__STACK.length>0)
If (Count parameters>=1)
$err:=$1
Else 
$err:=0
End if 

C_TEXT($vt_buffer)
$vt_buffer:=String(Current date;7)+" "+String(Current time;HH MM SS)+" "  // Add the date & time
$vt_buffer:=$vt_buffer+"[p"+String(Current process;"000")

  // Output the user name if it is defined.
C_TEXT(WEB_t_userName)
If (WEB_t_userName#"")
$vt_buffer:=$vt_buffer+", "+WEB_t_userName
End if 

$vt_buffer:=$vt_buffer+"]: ###### CALLING CHAIN DUMP"+Char(Carriage return)

  // Output the calling chain
C_LONGINT($i)
For ($i;0;__STACK.length-1)
$vt_buffer:=$vt_buffer+" Lvl "+String($i;"##00")+": "+("   "*($i))+__STACK[$i].tag
If (__STACK[$i].extraText#"")
$vt_buffer:=$vt_buffer+" -- "+__STACK[$i].extraText
End if 
$vt_buffer:=$vt_buffer+Char(Carriage return)
End for 

  // Output our ERROR # 
If ($err#0)
$vt_buffer:=$vt_buffer+" Lvl "+String(__STACK.length;"##00")+": "+("   "*(__STACK.length))+"above method triggered err #"+String($err)+Char(Carriage return)
End if 

LogNamed_AppendToFile ("Profiler Call Stack";$vt_buffer)
End if
