  //%attributes = {"invisible":true,"shared":true,"preemptive":"capable","folder":"Profiler Shared Methods","lang":"en"} comment added and reserved by 4D.
  // Profiler_START (methodName {;optionalText})
  // Profiler_START (text {;text})
  // 
  // DESCRIPTION
  //   Used to capture the start of a segment of code that
  //   is to be monitored.
  //   NOTE: Spaces in $1 & $2 are replaced by "_".
  //
C_TEXT($1;$nameToTrack)  // name to track
C_TEXT($2)  // OPTIONAL; optional text
  // ----------------------------------------------------
  // HISTORY
  //   Created by: DB (11/23/07)
  //   Mod: DB (10/29/2010) - Track start time
  //   Mod: DB (11/22/2010) - support performance tracking switch
  //   Mod: DB (01/20/2011) - Method stack always being tracked
  //   Mod: DB (05/22/2017) - Moved to v16 component
  // ----------------------------------------------------

ASSERT(Count parameters>0)
ASSERT(Count parameters<=2)
Init_ThreadSafe 

  // If this START is the first one for the stack and there is other stats, we want to reinit
If (Size of array(_ML_STACK_CallingStack)=0) & (Size of array(_ML_Perf_Method)>0)
Init_ThreadSafe (True)  // Force a Init
End if 

$nameToTrack:=Replace string($1;" ";"_")

  // Add our method to the call stack
APPEND TO ARRAY(_ML_STACK_CallingStack;$nameToTrack)
If (Count parameters=2)
APPEND TO ARRAY(_ML_STACK_ExtraText;Replace string($2;" ";"_"))
Else 
APPEND TO ARRAY(_ML_STACK_ExtraText;"")
End if 
APPEND TO ARRAY(_ML_STACK_msStart;Milliseconds)
APPEND TO ARRAY(_ML_STACK_wasteTime;0)

__incrementLevel:=__incrementLevel+1
