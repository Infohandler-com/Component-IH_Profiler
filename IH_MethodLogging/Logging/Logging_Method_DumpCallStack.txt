  //%attributes = {"invisible":true,"preemptive":"capable","lang":"en"} comment added and reserved by 4D.
  // Logging_Method_DumpCallStack ()
  // 
  // DESCRIPTION
  //   Dumps the call stack to the log
  //
C_LONGINT($1;$vl_ERROR)
  // ----------------------------------------------------
  // HISTORY
  //   Created by: DB (01/14/09)
  //   Mod: DB (05/22/2017) - Moved to v16 component
  // ----------------------------------------------------

Init_ThreadSafe 

If (Size of array(_ML_STACK_CallingStack)>0)
If (Count parameters>=1)
$vl_ERROR:=$1
Else 
$vl_ERROR:=0
End if 

C_TEXT($vt_buffer)
$vt_buffer:=String(Current date;7)+" "+String(Current time;HH MM SS)+" "  // Add the date & time
$vt_buffer:=$vt_buffer+"[p"+String(Current process;"000")

  // Output the user name if it is defined.
C_TEXT(WEB_t_userName)
If (WEB_t_userName#"")
$vt_buffer:=$vt_buffer+", "+WEB_t_userName
End if 

$vt_buffer:=$vt_buffer+"]: ###### CALLING CHAIN DUMP"+Char(Carriage return)

  // Output the calling chain
C_LONGINT($i)
For ($i;1;Size of array(_ML_STACK_CallingStack))
$vt_buffer:=$vt_buffer+" Lvl "+String($i;"##00")+": "+("   "*($i-1))+_ML_STACK_CallingStack{$i}
If (_ML_STACK_ExtraText{$i}#"")
$vt_buffer:=$vt_buffer+" -- "+_ML_STACK_ExtraText{$i}
End if 
$vt_buffer:=$vt_buffer+Char(Carriage return)
End for 

  // Output our ERROR # 
If ($vl_ERROR#0)
$vt_buffer:=$vt_buffer+" Lvl "+String(Size of array(_ML_STACK_CallingStack);"##00")+": "+("   "*(Size of array(_ML_STACK_CallingStack)))+"above method triggered err #"+String($vl_ERROR)+Char(Carriage return)
End if 

LogNamed_AppendToFile ("MethodStats";$vt_buffer)
End if
