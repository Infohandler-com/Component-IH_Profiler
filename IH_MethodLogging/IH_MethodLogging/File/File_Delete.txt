  // File_Delete (path to file)

  //   Deletes the document pass to it.

C_TEXT($1;$vt_fileName)
  // ----------------------------------------------------
  //   Created by: DB (07/25/07)
  // ----------------------------------------------------

$vt_fileName:=$1

If (False)
$CurrentTime:=Current time+?00:00:05?
Repeat 
IDLE
DELAY PROCESS(Current process;60)
Until (Current time>$CurrentTime)
End if 

If (File_DoesExist ($vt_fileName))
DELETE DOCUMENT($vt_fileName)
End if
