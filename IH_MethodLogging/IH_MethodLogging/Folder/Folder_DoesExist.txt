  // ----------------------------------------------------
  // METHOD: Folder_DoesExist (path to folder) : does exist
  // 
  // DESCRIPTION:
  //   Returns true if the folder exists
  // ----------------------------------------------------
  // PARAMETERS:
  //   $1: path to folder
  // RETURNS:
  //   $0: does exist
  // ----------------------------------------------------
  // MODIFICATION HISTORY:
  //   Added: DB (7/17/03 @ 15:46:39)
  // ----------------------------------------------------

C_BOOLEAN($0;$Folder_vb_doesExist)
$Folder_vb_doesExist:=False

If (Asserted(Count parameters=1))
C_TEXT($1;$Folder_vt_fullPath)
$Folder_vt_fullPath:=$1

If ($Folder_vt_fullPath#"")
C_LONGINT($Folder_vl_pathNameTestCode)
$Folder_pathNameTestCode:=Test path name($Folder_vt_fullPath)
If ($Folder_pathNameTestCode=Is a folder)
$Folder_vb_doesExist:=True
End if 
End if 
End if   // ASSERT

$0:=$Folder_vb_doesExist
