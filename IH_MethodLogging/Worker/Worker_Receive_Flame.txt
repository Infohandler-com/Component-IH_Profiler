ASSERT(Count parameters=1)
$obj:=$1

OnErr_Install_Handler ("OnErr_GENERIC")

Init_ThreadSafe 

ARRAY TEXT($at_method;0)
ARRAY LONGINT($al_minTime;0)
ARRAY LONGINT($al_maxTime;0)
ARRAY REAL($ar_totalTime;0)
ARRAY LONGINT($al_count;0)

OnErr_ClearError 
If (OB Is defined($obj;"methods"))
OB GET ARRAY($obj;"methods";$at_method)
OB GET ARRAY($obj;"minTime";$al_minTime)
OB GET ARRAY($obj;"maxTime";$al_maxTime)
OB GET ARRAY($obj;"totalTime";$ar_totalTime)
OB GET ARRAY($obj;"count";$al_count)
Else 
EXECUTE METHOD("SLACK_SendWarning";*;Current method name+": encountered an issue with arrays")
Log_ERR_CRITICAL ("IH_Profiler Component Error --> "+Current method name+" $obj is--> "+JSON Stringify($obj;*);"On Err Trigger")
LogNamed_AppendToFile ("On Err Trigger";"IH_Profiler Component Error --> "+Current method name+" $obj is--> "+JSON Stringify($obj;*);"On Err Trigger")
End if 

If (OnErr_GetLastError =0)

C_LONGINT($pos;$i)
For ($i;1;Size of array($at_method))
$pos:=Find in array(_ML_Flame_Method;$at_method{$i})
If ($pos<1)  // not found
APPEND TO ARRAY(_ML_Flame_Method;$at_method{$i})
APPEND TO ARRAY(_ML_Flame_minTime;$al_minTime{$i})
APPEND TO ARRAY(_ML_Flame_maxTime;$al_maxTime{$i})
APPEND TO ARRAY(_ML_Flame_totalTime;$ar_totalTime{$i})
APPEND TO ARRAY(_ML_Flame_count;$al_count{$i})
Else 

If (_ML_Flame_minTime{$pos}>$al_minTime{$i})
_ML_Flame_minTime{$pos}:=$al_minTime{$i}
End if 

If (_ML_Flame_maxTime{$pos}<$al_maxTime{$i})
_ML_Flame_maxTime{$pos}:=$al_maxTime{$i}
End if 

_ML_Flame_totalTime{$pos}:=_ML_Flame_totalTime{$pos}+$ar_totalTime{$i}
_ML_Flame_count{$pos}:=_ML_Flame_count{$pos}+$al_count{$i}
End if 
End for 

Else 

EXECUTE METHOD("SLACK_SendWarning";*;Current method name+": encountered an issue with arrays")
Log_ERR_CRITICAL ("IH_Profiler Component Error --> "+Current method name+" $obj is--> "+JSON Stringify($obj;*);"On Err Trigger")
LogNamed_AppendToFile ("On Err Trigger";"IH_Profiler Component Error --> "+Current method name+" $obj is--> "+JSON Stringify($obj;*);"On Err Trigger")

End if
