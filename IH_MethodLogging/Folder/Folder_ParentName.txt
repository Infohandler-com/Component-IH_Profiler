  //Method: Folder_ParentName
  //Created by: Peter, Feb, 2000 11:30
  //Called from many places including StartWindows, StartMacintosh
  //Returns the Parent Name of the file pathname we pass in
  //
  // Last Modified by: DB (7/25/03 @ 09:28:20) - added alternate seperator

C_TEXT($0;$HFS_ParentName)
$HFS_ParentName:=""

If (Asserted((Count parameters>=1) & (Count parameters<=2)))
C_TEXT($1;$HFS_FullPath)
C_TEXT($2)
C_TEXT($HFS_AltSeperator)
$HFS_FullPath:=$1
If (Count parameters=2)
$HFS_AltSeperator:=$2
Else 
$HFS_AltSeperator:=Folder separator
End if 

C_LONGINT($i)
For ($i;Length($HFS_FullPath);1;-1)
If ($HFS_FullPath[[$i]]=$HFS_AltSeperator) & ($i#Length($HFS_FullPath))
$HFS_ParentName:=Substring($HFS_FullPath;1;$i)
$i:=0  //end loop
End if 
End for 
End if   // ASSERT

$0:=$HFS_ParentName
