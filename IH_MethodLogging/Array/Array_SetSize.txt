  //%attributes = {"invisible":true,"preemptive":"capable","lang":"en"} comment added and reserved by 4D.
  // (PM) Array_SetSize
  // Changes the size of one or multiple arrays
  // $1 = Size
  // $2 etc. = Pointers to arrays

C_LONGINT($1;$size;$param;$currentSize)
C_POINTER(${2};$array)

C_TEXT($vt_prevOnErr)
$vt_prevOnErr:=Method called on error

ON ERR CALL("OnErr_GENERIC")
OnErr_ClearError 

$size:=$1
For ($param;2;Count parameters)

$array:=${$param}
$currentSize:=Size of array($array->)
Case of 
: ($currentSize<$size)
INSERT IN ARRAY($array->;$currentSize+1;$size-$currentSize)
: ($currentSize>$size)
DELETE FROM ARRAY($array->;$size+1;$currentSize-$size)
End case 

End for 

ON ERR CALL($vt_prevOnErr)
