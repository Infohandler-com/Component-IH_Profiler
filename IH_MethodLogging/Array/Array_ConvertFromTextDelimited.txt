  //%attributes = {"invisible":true,"preemptive":"capable","lang":"en"} comment added and reserved by 4D.
  // Array_ConvertFromTextDelimited (ArrayPtr, srcText{; delimiter})
  // Array_ConvertFromTextDelimited (pointer; text{; text})
  //
  // DESCRIPTION
  //   Converts a delimited text string into values
  //   in the passed text array.  The delimiter defaults to "," if not supplied.
  //
C_POINTER($1;$vp_arrayPtr)
C_TEXT($2;$vt_srcTxt)
C_TEXT($3;$theDelimiter)  // OPTIONAL
  // ----------------------------------------------------
  // HISTORY
  //   Created by: Dani Beaubien (04/19/07)
  //   Mod: DB (09/25/2012) - Fixed bug if last character of srcTxt is the delimiter
  //   Mod: DB (09/25/2012) - Removed code that trimed extra spaces
  // ----------------------------------------------------

ASSERT(Count parameters>=2)
ASSERT(Count parameters<=3)

$vp_arrayPtr:=$1
$vt_srcTxt:=$2
If (Count parameters=3)
$theDelimiter:=$3
Else 
$theDelimiter:=","
End if 

Array_SetSize (0;$vp_arrayPtr)

C_LONGINT($vl_delSize;$pos)
$vl_delSize:=Length($theDelimiter)

If ($vt_srcTxt#"")
$pos:=Position($theDelimiter;$vt_srcTxt;*)
While ($pos>0)
APPEND TO ARRAY($vp_arrayPtr->;Substring($vt_srcTxt;1;$pos-1))
$vt_srcTxt:=Substring($vt_srcTxt;$pos+$vl_delSize)

$pos:=Position($theDelimiter;$vt_srcTxt;*)
End while 

  //   Mod: DB (09/25/2012) - Add the an element for the last line.
APPEND TO ARRAY($vp_arrayPtr->;$vt_srcTxt)
End if
